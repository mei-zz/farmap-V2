import requests
from PIL import Image
from io import BytesIO
import os
from dotenv import load_dotenv

# 加载环境变量
load_dotenv()

def identify_single_image(image_url: str, api_key: str) -> dict:
    """识别单张图片并返回结果"""
    try:
        # 1. 下载图片
        print(f"正在下载图片: {image_url}")
        response = requests.get(image_url, timeout=30)
        response.raise_for_status()
        
        # 检查响应内容
        if len(response.content) == 0:
            print("图片内容为空")
            return {"success": False, "error": "图片内容为空"}
            
        # 2. 调用PlantNet API
        url = f"https://my-api.plantnet.org/v2/identify/all?api-key={api_key}"
        files = {"images": ("image.jpg", response.content, "image/jpeg")}
        data = {"organs": "auto"}
        
        print("正在调用PlantNet API...")
        api_response = requests.post(url, files=files, data=data, timeout=30)
        
        if api_response.status_code != 200:
            print(f"API调用失败，响应内容: {api_response.text}")
            return {"success": False, "error": f"API调用失败: {api_response.text}"}
            
        result = api_response.json()
        
        # 3. 提取识别结果
        if result.get("results") and len(result["results"]) > 0:
            top_result = result["results"][0]
            species_info = top_result["species"]
            scientific_name = species_info["scientificName"]
            confidence = top_result["score"]
            
            return {
                "success": True,
                "scientific_name": scientific_name,
                "confidence": confidence,
                "full_response": result
            }
        else:
            return {"success": False, "error": "未识别到物种"}
            
    except Exception as e:
        print(f"识别图片时发生错误: {str(e)}")
        return {"success": False, "error": str(e)}

def debug_plantnet_api():
    """调试PlantNet API调用，支持多张图片"""
    # 获取API密钥
    api_key = os.getenv("PLANTNET_API_KEY")
    if not api_key:
        print("错误: 未配置PLANTNET_API_KEY环境变量")
        return
    
    # 测试图片URL列表（使用一些已知有效的蓝莓图片URL）
    test_image_urls = [
        "https://preview.qiantucdn.com/freepik/free-vector/65656c5557c17.jpg!w1024_new_small_1",
        "https://ts1.tc.mm.bing.net/th/id/R-C.420fab226d4322a90c403d679a153607?rik=oir2XJMmNVviYA&riu=http%3a%2f%2fn.sinaimg.cn%2fsinakd20200604s%2f3%2fw1125h478%2f20200604%2fb6fa-iumkapx2807772.jpg&ehk=0W6XLYY9kPpBjfc1xnefBHSgm%2fDkW99wXlyAz4ynp%2bQ%3d&risl=&pid=ImgRaw&r=0",
        "https://img95.699pic.com/photo/30797/5512.jpg_wh300.jpg"
    ]
    
    print(f"API密钥: {api_key[:5]}...")
    print(f"测试图片数量: {len(test_image_urls)}")
    
    # 存储所有识别结果
    all_results = []
    
    # 逐个识别每张图片
    for i, image_url in enumerate(test_image_urls, 1):
        print(f"\n--- 正在处理第{i}张图片 ---")
        result = identify_single_image(image_url, api_key)
        result["image_index"] = i
        result["image_url"] = image_url
        all_results.append(result)
        
        if result["success"]:
            print(f"第{i}张图片识别成功:")
            print(f"  科学名称: {result['scientific_name']}")
            print(f"  置信度: {result['confidence']:.2f}")
        else:
            print(f"第{i}张图片识别失败: {result['error']}")
    
    # 输出汇总结果
    print("\n=== 汇总结果 ===")
    successful_identifications = [r for r in all_results if r["success"]]
    
    if len(successful_identifications) == 0:
        print("所有图片识别都失败了")
        return
    
    # 检查是否所有成功识别的图片都属于同一物种
    if len(successful_identifications) > 1:
        first_species = successful_identifications[0]["scientific_name"]
        all_same_species = all(r["scientific_name"] == first_species for r in successful_identifications)
        
        if all_same_species:
            print(f"✓ 所有成功识别的图片都属于同一物种: {first_species}")
        else:
            print("✗ 成功识别的图片属于不同物种:")
            for result in successful_identifications:
                print(f"  第{result['image_index']}张: {result['scientific_name']} (置信度: {result['confidence']:.2f})")
    elif len(successful_identifications) == 1:
        print(f"只有1张图片识别成功: {successful_identifications[0]['scientific_name']} (置信度: {successful_identifications[0]['confidence']:.2f})")
    
    # 计算平均置信度
    if successful_identifications:
        avg_confidence = sum(r["confidence"] for r in successful_identifications) / len(successful_identifications)
        print(f"平均置信度: {avg_confidence:.2f}")

if __name__ == "__main__":
    debug_plantnet_api()
